    def _deep_merge(self, base: Dict, update: Dict) -> Dict:
        """Deep merge two dictionaries."""
        result = base.copy()
        for key, value in update.items():
            if key in result and isinstance(result[key], dict) and isinstance(value, dict):
                result[key] = self._deep_merge(result[key], value)
            else:
                result[key] = value
        return result

class KnowledgeNode(db.Model):
    """Knowledge graph nodes for The Hoard memory system"""
    __tablename__ = 'knowledge_nodes'
    
    id = Column(Integer, primary_key=True)
    content = Column(Text, nullable=False)
    content_hash = Column(String(64), nullable=False, unique=True, index=True)
    node_type = Column(String(50), nullable=False)
    metadata = Column(JSON, nullable=True)
    embedding = Column(JSON, nullable=True)  # Vector embeddings as JSON
    created_at = Column(DateTime, default=datetime.utcnow, nullable=False)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    accessed_count = Column(Integer, default=0)
    last_accessed = Column(DateTime, nullable=True)
    importance_score = Column(Float, default=0.0)
    
    # Relationships
    outgoing_edges = relationship("KnowledgeEdge", foreign_keys="KnowledgeEdge.source_id", back_populates="source")
    incoming_edges = relationship("KnowledgeEdge", foreign_keys="KnowledgeEdge.target_id", back_populates="target")
    
    def to_dict(self):
        return {
            'id': self.id,
            'content': self.content,
            'node_type': self.node_type,
            'metadata': self.metadata,
            'created_at': self.created_at.isoformat(),
            'updated_at': self.updated_at.isoformat(),
            'accessed_count': self.accessed_count,
            'last_accessed': self.last_accessed.isoformat() if self.last_accessed else None,
            'importance_score': self.importance_score
        }

    """Knowledge graph edges connecting nodes"""
    __tablename__ = 'knowledge_edges'

    weight = Column(Float, default=1.0)
