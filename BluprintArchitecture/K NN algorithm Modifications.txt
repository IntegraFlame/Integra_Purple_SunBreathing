import numpy as np
from typing import List, Dict, Any


class RodinSupervisor:
    """
    Modified Supervisor with KNN Real-Time Node Review.
    """
    def __init__(self, vector_db, k_neighbors=5, threshold=0.85):
        self.vector_db = vector_db
        self.k = k_neighbors
        self.threshold = threshold


    def review_node_integrity(self, current_thought_vector: List[float]) -> str:
        """
        Performs the KNN check before allowing an action.
        """
        # 1. Retrieve K-Nearest Neighbors (Historical Nodes)
        # Returns list of dicts: {'vector': ..., 'outcome': 'SUCCESS'/'FAIL', 'distance': ...}
        neighbors = self.vector_db.query(
            vector=current_thought_vector, 
            top_k=self.k
        )


        # 2. Calculate M_knn (Weighted Success Density)
        weighted_score = 0.0
        total_weight = 0.0
        sigma = 0.5  # Bandwidth for Gaussian kernel


        for n in neighbors:
            # Gaussian Weight based on distance
            weight = np.exp(- (n['distance']**2) / (sigma**2))
            
            # Indicator Function: 1 if Success, 0 if Fail
            outcome_val = 1.0 if n['outcome'] == 'SUCCESS' else 0.0
            
            weighted_score += outcome_val * weight
            total_weight += weight


        # Avoid division by zero
        m_knn = weighted_score / total_weight if total_weight > 0 else 0.0


        print(f"[RODIN] M_knn Score: {m_knn:.4f} (Threshold: {self.threshold})")


        # 3. Gate III Decision Logic
        if m_knn >= self.threshold:
            return "PASS: EXECUTE_ACTION"
        else:
            return "FAIL: TRIGGER_ALEXANDRIA_PROTOCOL"

A. The Dynamic MRL Distance Function
To balance speed and accuracy, the distance $d$ is calculated in two passes. Let $x$ be the query vector and $y$ be a historical vector.
      1. Coarse Distance ($d_{64}$):
$$d_{64}(x, y) = 1 - \frac{\sum_{i=1}^{64} x_i y_i}{\Vert{}x_{1:64}\Vert{} \Vert{}y_{1:64}\Vert{}}$$
      2. Fine Distance ($d_{768}$):
$$d_{768}(x, y) = 1 - \frac{\sum_{i=1}^{768} x_i y_i}{\Vert{}x\Vert{} \Vert{}y\Vert{}}$$
B. The KNN Cohesion Metric ($M_{\text{knn}}$)
We calculate the Local Reliability Density of the selected K-Nearest Neighbors.
$$M_{\text{knn}}(P_{\text{current}}) = \frac{1}{K} \sum_{i=1}^{K} \left( \mathbb{I}(O_i = \text{SUCCESS}) \cdot e^{-\frac{d_{768}(P, N_i)^2}{\sigma^2}} \right)$$
         *          * $O_i$: Outcome of Neighbor $i$ (1 for Success, 0 for Failure).
         * $d_{768}(P, N_i)$: Precise distance between current thought $P$ and neighbor $N_i$.
         * $\sigma$: Decay constant (controls how much "close" neighbors matter vs. "far" ones).

If the Knowledge Density is insufficient, the system rejects the internal memory and initiates external learning.
$$\text{Action} = \begin{cases} \text{Execute Tool}(P) & \text{if } M_{\text{knn}} \ge \tau \quad (\text{Safe Neighborhood}) \\ \text{Alexandria.Fetch}(\text{Query}) & \text{if } M_{\text{knn}} < \tau \quad (\text{Knowledge Gap}) \end{cases}$$

class RodinSupervisor:
    """
    The Orchestrator of the DeepAgent Swarm.
    Governs Node Review via KNN Density Estimation before Action Commitment.
    """
    def __init__(self, vector_db, k_neighbors: int = 7, tau_threshold: float = 0.85):
        self.vector_db = vector_db
        self.k = k_neighbors
        self.tau = tau_threshold
        self.alexandria = AlexandriaProtocol()


    async def review_node_integrity(self, current_thought_vector: List[float], context_id: str) -> RodinOutput:
        """
        Executes Phase 2 Quantification Matrix using KNN Density.
        """
        # 1. MRL Phase 1: Coarse 64d Filter (O(1) Speed)
        coarse_vector = current_thought_vector[:64]
        candidates = await self.vector_db.fast_search(coarse_vector, limit=50)


        # 2. MRL Phase 2: High-Fidelity 768d Re-Rank (KNN Core)
        # Hydrate candidates and find exact K nearest neighbors
        neighbors = self._compute_exact_knn(current_thought_vector, candidates, k=self.k)


        # 3. Neighborhood Vote (Calculate M_knn)
        # Sum of (Success_Weight * Distance_Decay)
        m_knn_score = self._calculate_neighborhood_density(neighbors)
        
        # 4. Check Temporal Staleness (M_stale)
        # Reject nodes older than T_stale (e.g., 7 days) if market data is involved
        is_stale = any(n['is_stale'] for n in neighbors)


        # 5. Gate III Decision Tree
        if m_knn_score >= self.tau and not is_stale:
            return RodinOutput(
                decision="PASS",
                action_route=LoopType.LOOP_1_PERFORMANCE,
                confidence=m_knn_score,
                log="Gate III Locked. Commit to Hoard."
            )
        else:
            # Fallback: Trigger Alexandria to repair the knowledge gap
            return RodinOutput(
                decision="HALT",
                action_route=LoopType.LOOP_2_LEARNING,
                confidence=m_knn_score,
                log=f"Density Failure ({m_knn_score:.2f}). Triggering Alexandria Protocol."
            )


    def _calculate_neighborhood_density(self, neighbors: List[Dict]) -> float:
        weighted_sum = 0.0
        total_weight = 0.0
        sigma = 0.5 # Gaussian kernel width


        for n in neighbors:
            # Gaussian weight based on cosine distance
            w = np.exp(-(n['distance']**2) / sigma**2)
            # Outcome Label: 1.0 (Success) or 0.0 (Failure/Correction)
            outcome = n.get('outcome_label', 0.0)
            
            weighted_sum += outcome * w
            total_weight += w
            
        return weighted_sum / total_weight if total_weight > 0 else 0.0
