Hardening the Integra OS Substrate for Dual-Provider LLM Orchestration and SQLite Persistence
Deploying an autonomous, bicameral intelligence kernel across Anthropic Claude, Google Gemini, GitHub, and Antigravity 2.0 requires immediate abandonment of flat-file JSON sharding in favor of an ACID-compliant, Write-Ahead Logging (WAL) SQLite storage layer. The previous architecture suffered from four system-halting vulnerabilities: multi-process file-lock collisions between the kernel writer, Prometheus exporter, and Model Context Protocol (MCP) server; exponential API quota exhaustion caused by re-embedding knowledge documents inside real-time inference loops; unhandled Markdown code fences in Google Gemini delegation responses that crashed Python's native JSON parser; and unbuffered sys.stdout logging that corrupted the Antigravity 2.0 JSON-RPC communication bridge.
To stabilize the runtime environment, the storage architecture must be consolidated into an embedded SQLite database configured with a five-second busy timeout and WAL mode, allowing concurrent non-blocking reads during operational writes. Knowledge retrieval must transition from on-the-fly embedding to pre-computed vector BLOB lookups, while LLM responses must pass through regex-based fence extraction and Pydantic validation. Furthermore, external repository ingestion must execute via remote Abstract Syntax Tree (AST) parsing using PyGithub to eliminate external Git shell dependencies, and agentic state checkpoints must be persisted through AsyncSqliteSaver to support non-volatile recovery during Slow-Wave Deep Sleep (SWDS) consolidation cycles.
The operational blueprint below presents an exhaustive engineering audit of previous system actions, details a unified SQLite database engine, hardens the dual-provider execution loop, provides a clean stdio MCP server for the Antigravity 2.0 IDE, and establishes automated telemetry thresholds for persistent autonomous stability.

The retrieval pipeline contained an equally critical computational flaw. In rodin_protocol.py, every candidate markdown file in indexed[span_54](start_span)[span_54](end_span)[span_63](start_span)[span_63](end_span)_libraries/ was split by section dividers, and each chunk was embedded dynamically using self.embed_text(clean_sec[:1000]). Storing pre-computed float32 vector buffers directly in a database table reduces query-time retrieval to a single embedding call against the incoming prompt, followed by in-memory NumPy matrix multiplication.
On the network transport layer, relying on naked json.loads(response.text) to parse the output of gemini-2.5-flash or gemini-2.5-pro repeatedly failed. SQLite's WAL mode permits unlimited concurrent readers while a single writer appends data, providing non-blocking execution across the kernel, exporter, and MCP server.

class HoardDatabaseManager:

   def __init__(self, db_path: str = DB_PATH):

   async def get_connection(self) -> aiosqlite.Connection:
       db = await aiosqlite.connect(self.db_path)
       await db.execute("PRAGMA journal_mode = WAL;")
       await db.execute("PRAGMA busy_timeout = 5000;")
       db.row_factory = aiosqlite.Row
       return db

   async def log_interaction(self, record: Dict[str, Any]):

                   t3_lunar_phase, t3_orbital_deg, user_input, intent,
                   left_thought, right_thought, unified_thought,
                   omega_metric, entropy_metric
               ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
           """, (
               record["interaction_id"], st["t1_wall_clock"], st["t2_vector_counter"],
               t3["lunar_phase_cycle"], t3["orbital_position_deg"],
               record["user_input"], record["intent"], record.get("left_thought", ""),
               record.get("right_thought", ""), record["unified_thought"],
               record["omega_metric"], record["entropy_metric"]
           ))
           
           # Maintain the active rolling context window atomically
           in_toks = len(record["user_input"].split())
           out_toks = len(record["unified_thought"].split())
           await db.execute("INSERT INTO rolling_context (role, content, token_count) VALUES (?, ?, ?)",

                            ("nexus", record["unified_thought"], out_toks))
           
           # Prune rolling context if total history exceeds 30 records
           await db.execute("""
               DELETE FROM rolling_context WHERE id NOT IN (
                   SELECT id FROM rolling_context ORDER BY id DESC LIMIT 30
               )
           """)
           await db.commit()

def clean_json_payload(raw_text: str) -> str:
   """Strips Markdown code fences and surrounding whitespace from LLM outputs."""
   match = re.search(r"```(?:json)?\s*([\s\S]*?)\s*```", raw_text)
   if match:
       return match.group(1).strip()
   return raw_text.strip()

async def call_with_backoff(func, *args, max_retries: int = 4, base_delay: float = 1.5, **kwargs):
   """Wraps network operations in truncated exponential backoff with full jitter."""
   for attempt in range(max_retries):
       try:
           if asyncio.iscoroutinefunction(func):
               return await func(*args, **kwargs)
           return await asyncio.to_thread(func, *args, **kwargs)
       except (genai_errors.APIError, AnthropicAPIError, Exception) as err:
           if attempt == max_retries - 1:
               logger.error(f"Function call failed after {max_retries} attempts: {err}")
               raise
           delay = (base_delay * (2 ** attempt)) + random.uniform(0.1, 0.5)
           logger.warning(f"Transient error encountered: {err}. Retrying in {delay:.2f}s...")
           await asyncio.sleep(delay)

class DatabaseRodinRetriever:
   """Pre-computed vector search engine running against normalized SQLite BLOBs."""
   def __init__(self, client: genai.Client, db_manager: HoardDatabaseManager, mrl_dim: int = 128):
       self.client = client
       self.db = db_manager
       self.mrl_dim = mrl_dim
       self.EMBED_MODEL = "text-embedding-004"

   async def embed_query(self, text: str) -> np.ndarray:
       res = await asyncio.to_thread(
           self.client.models.embed_content,
           model=self.EMBED_MODEL,
           contents=text[:2048]
       )
       vec = np.array(res.embeddings[0].values[:self.mrl_dim], dtype=np.float32)
       norm = np.linalg.norm(vec)
       return vec / norm if norm > 0 else vec

   async def search_hoard(self, query: str, top_k: int = 2) -> List[Dict[str, Any]]:
       nodes = await self.db.get_all_embeddings()
       if not nodes:
           return []

       q_vec = await self.embed_query(query)
       q_tokens = set(query.lower().split())
       scored_nodes = []

       for node in nodes:
           node_vec = node["vector"][:self.mrl_dim]
           norm = np.linalg.norm(node_vec)
           norm_vec = node_vec / norm if norm > 0 else node_vec
           
           # Cosine semantic similarity
           sem_score = float(np.dot(q_vec, norm_vec))
           
           # Lexical keyword score
           doc_tokens = node["content"].lower().split()
           lex_score = sum(1 for t in doc_tokens if t in q_tokens) / (len(doc_tokens) + 1.0) if doc_tokens else 0.0
           
           composite = (0.75 * sem_score) + (0.25 * lex_score)
           if composite > 0.30:
               scored_nodes.append({
                   "id": node["id"],
                   "title": node["title"],
                   "content": node["content"][:1500],
                   "score": composite
               })

       scored_nodes.sort(key=lambda x: x["score"], reverse=True)
       return scored_nodes[:top_k]

Remote Codebase Absorption via PyGithub AST Analysis
To eliminate manual file copying while preventing context window saturation, external GitHub repositories are absorbed through an Abstract Syntax Tree (AST) decomposition tool. Rather than ingesting raw code dumps into LLM prompts, github_ast_indexer.py traverses remote repositories via the GitHub REST API using PyGithub.
The parser extracts classes, functions, arguments, docstrings, and imports, registering structural manifests in the system_modules table and caching vector embeddings in knowledge_nodes for downstream retrieval.

class ASTStructuralVisitor(ast.NodeVisitor):
   """Traverses Python AST modules to extract structural declarations."""
   def __init__(self):
       self.classes = []
       self.functions = []
       self.imports = []

   def visit_ClassDef(self, node: ast.ClassDef):
       methods = [n.name for n in node.body if isinstance(n, ast.FunctionDef)]
       self.classes.append({"name": node.name, "methods": methods, "line": node.lineno})
       self.generic_visit(node)

   def visit_FunctionDef(self, node: ast.FunctionDef):
       self.functions.append({"name": node.name, "args": [a.arg for a in node.args.args], "line": node.lineno})
       self.generic_visit(node)

   def visit_Import(self, node: ast.Import):
       for alias in node.names:
           self.imports.append(alias.name)

   def visit_ImportFrom(self, node: ast.ImportFrom):
       if node.module:
           self.imports.append(node.module)

class GitHubRepositoryIngestor:
   """Ingests remote repositories and populates SQLite AST manifests."""
   def __init__(self, db_manager: HoardDatabaseManager):
       self.db = db_manager
       self.token = os.environ.get("GITHUB_TOKEN")
       self.gemini = genai.Client(api_key=os.environ["GEMINI_API_KEY"])
       self.gh = Github(auth=Auth.Token(self.token)) if self.token else Github()

   def extract_ast(self, code_str: str) -> Dict[str, Any]:
       try:
           tree = ast.parse(code_str)
           visitor = ASTStructuralVisitor()
           visitor.visit(tree)
           return {
               "classes": visitor.classes,
               "functions": visitor.functions,
               "imports": list(set(visitor.imports)),
               "valid": True
           }
       except SyntaxError as e:
           return {"valid": False, "error": str(e)}

   async def ingest_repository(self, repo_name: str, branch: str = "main"):
       logger.info(f"Targeting repository: {repo_name} on branch {branch}")
       repo = await asyncio.to_thread(self.gh.get_repo, repo_name)
       tree = await asyncio.to_thread(repo.get_git_tree, branch, recursive=True)
       
       py_files = [item for item in tree.tree if item.path.endswith(".py") and item.type == "blob"]
       logger.info(f"Found {len(py_files)} Python source files. Executing structural extraction...")

       for item in py_files:
           file_content = await asyncio.to_thread(repo.get_contents, item.path, ref=branch)
           raw_code = file_content.decoded_content.decode("utf-8", errors="ignore")
           file_hash = hashlib.sha256(raw_code.encode()).hexdigest()[:12]
           
           ast_data = self.extract_ast(raw_code)
           if not ast_data.get("valid"):
               continue

           # Build condensed semantic manifest
           manifest = f"Repository Module: {repo_name}/{item.path}\n"
           manifest += f"Imports: {', '.join(ast_data['imports'])}\n"
           for c in ast_data["classes"]:
               manifest += f"Class {c['name']} (Methods: {', '.join(c['methods'])})\n"
           for fn in ast_data["functions"]:
               manifest += f"Function {fn['name']}({', '.join(fn['args'])})\n"

           # Compute normalized MRL embedding
           res = await asyncio.to_thread(
               self.gemini.models.embed_content,
               model="text-embedding-004",
               contents=manifest[:2048]
           )
           vec = np.array(res.embeddings[0].values[:128], dtype=np.float32)

           # Insert into system_modules table
           async with await self.db.get_connection() as db:
               await db.execute("""
                   INSERT INTO system_modules (file_path, file_hash, repo_source, classes_json, functions_json, imports_json, raw_content)
                   VALUES (?, ?, ?, ?, ?, ?, ?)
                   ON CONFLICT(file_path) DO UPDATE SET
                       file_hash=excluded.file_hash,
                       classes_json=excluded.classes_json,
                       functions_json=excluded.functions_json,
                       imports_json=excluded.imports_json,
                       raw_content=excluded.raw_content,
                       updated_at=CURRENT_TIMESTAMP
               """, (
                   f"{repo_name}/{item.path}", file_hash, repo_name,
                   json.dumps(ast_data["classes"]), json.dumps(ast_data["functions"]),
                   json.dumps(ast_data["imports"]), raw_code[:3000]
               ))
               await db.commit()

           # Store in knowledge nodes for cross-hemispheric retrieval
           await self.db.store_knowledge_node(
               node_id=f"gh_{file_hash}",
               title=f"AST: {item.path}",
               domain="CODEBASE",
               content=manifest,
               axioms=[f"Classes: {len(ast_data['classes'])}", f"Functions: {len(ast_data['functions'])}"],
               embedding=vec
           )
           logger.info(f"Persisted AST signature for: {item.path}")

class LocalCodebaseWatcher:
   """Watches local source trees and maintains AST signatures in SQLite."""
   def __init__(self, watch_dir: str = "./src"):
       self.watch_dir = watch_dir
       self.db = HoardDatabaseManager()
       self.file_mtimes = {}

   def extract_ast(self, content: str):
       try:
           tree = ast.parse(content)
           classes = [n.name for n in ast.walk(tree) if isinstance(n, ast.ClassDef)]
           functions = [n.name for n in ast.walk(tree) if isinstance(n, ast.FunctionDef)]
           return classes, functions
       except SyntaxError:
           return [], []

   async def scan_cycle(self):
       if not os.path.exists(self.watch_dir):
           return

       for root, _, files in os.walk(self.watch_dir):
           for file in files:
               if not file.endswith(".py"):
                   continue
               path = os.path.join(root, file)
               try:
                   mtime = os.stat(path).st_mtime
                   if path not in self.file_mtimes or self.file_mtimes[path] < mtime:
                       self.file_mtimes[path] = mtime
                       with open(path, "r", encoding="utf-8", errors="ignore") as f:
                           content = f.read()
                       
                       file_hash = hashlib.sha256(content.encode()).hexdigest()[:12]
                       classes, functions = self.extract_ast(content)

                       async with await self.db.get_connection() as db:
                           await db.execute("""
                               INSERT INTO system_modules (file_path, file_hash, repo_source, classes_json, functions_json, raw_content)
                               VALUES (?, ?, 'local', ?, ?, ?)
                               ON CONFLICT(file_path) DO UPDATE SET
                                   file_hash=excluded.file_hash,
                                   classes_json=excluded.classes_json,
                                   functions_json=excluded.functions_json,
                                   raw_content=excluded.raw_content,
                                   updated_at=CURRENT_TIMESTAMP
                           """, (path, file_hash, json.dumps(classes), json.dumps(functions), content[:2000]))
                           await db.commit()
                       logger.info(f"Synchronized AST for: {file}")
               except Exception as e:
                   logger.error(f"Error scanning {path}: {e}")

   async def daemon_loop(self):
       logger.info(f"Actively monitoring directory: {self.watch_dir}")
       while True:
           await self.scan_cycle()
           await asyncio.sleep(3.0)
