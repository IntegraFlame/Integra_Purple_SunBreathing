class BrailleCodec:
    BRAILLE_BASE = 0x2800

    @classmethod
    def byte_to_braille(cls, byte_val: int) -> str:
        return chr(cls.BRAILLE_BASE + (byte_val & 0xFF))

    @classmethod
    def braille_to_byte(cls, braille_char: str) -> int:
        return ord(braille_char) - cls.BRAILLE_BASE

    @classmethod
    def bytes_to_braille_str(cls, data: bytes) -> str:
        return "".join(cls.byte_to_braille(b) for b in data)

    @staticmethod
    def interleave_morton_16bit(y_lat_idx: int, x_lon_idx: int) -> int:
        y = max(0, min(0xFFFF, y_lat_idx))
        x = max(0, min(0xFFFF, x_lon_idx))
        morton_32 = 0
        for cell_idx in range(4):
            byte_val = 0
            for bit_step in range(4):
                shift = 15 - (cell_idx * 4 + bit_step)
                y_bit = (y >> shift) & 1
                x_bit = (x >> shift) & 1
                offset = bit_step * 2
                byte_val |= (y_bit << offset) | (x_bit << (offset + 1))
            morton_32 = (morton_32 << 8) | byte_val
        return morton_32
